> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 `docs/cpu/armv8-isa/` 全部 372 个指令页建立「gem5 O3 微架构链路」图谱（YAML 单一数据源 + 生成器 + 校验器 + clangd 工具），产出每指令一页、15 个分类索引、O3 总链路图、指令×注入器命中矩阵。

## 已核实的关键事实（计划依据，全部有 file:line 证据）

1. **语料**：387 md = 15 索引页 + 372 指令页。分类计数：Load-Store 123 · Other 81 · Arithmetic 33 · Bit-Manipulation 25 · Logical 23 · System 16 · Barrier-Hint 16 · Branch 13 · Pointer-Authentication 11 · Compare 11 · Conditional-Select 7 · Atomic 7 · PC-Relative 2 · Cryptographic 2 · Count 2。
2. **解码真源**是 `CHAOS/gem5/src/arch/arm/isa/formats/aarch64.isa`（3875 行，`decoder/aarch64.isa` 是 45 行 stub）。顶层分发：decodeDataProcImm / decodeBranchExcSys / decodeLoadsStores / decodeDataProcReg / decodeFpAdvSIMD。
3. **生成物**：`build/ARM/arch/arm/generated/`：`decoder-ns.hh.inc`（3790 个类声明，含 ARM/Thumb）、`decoder-ns.cc.inc`（构造函数首字符串字面量 = 助记符，如 `AddXImm...("add",...)`）、`decode-method.cc.inc`（顶层 switch）。`inst-constrs-{1,2,3}.cc` 是 include stub。
4. **类名规律**：`AddXImm`（ADD imm）/ `AddXSRegCc`（ADDS shifted reg）/ `LDRX64_IMM`（后缀 _IMM/_PRE/_POST/_REG/_LIT = 寻址，Cc = 置标志，B/H/W/X = 访问宽度）；LDP/STP 是宏展开单类 `LdpStp`（动态助记符）。
5. **已知未实现**：MTE 全部（CPY*/SETG*/STG*/LDGM/IRG/GMI——源码零匹配）absent；SYS/SYSL absent（落到 Unknown64）；BTI/ESB/PSB CSYNC/TSB CSYNC/CSDB = WarnUnimplemented（解码为 warn-nop）。
6. **O3 标准节点锚点**（families.yaml 种子，写入前用 `sed -n 'Np'` 复核行内容）：
   - FETCH: `src/cpu/o3/fetch.cc:1059`（Fetch::fetch）
   - DECODE: `src/cpu/o3/decode.cc:636`（Decode::decodeInsts）
   - RENAME: `src/cpu/o3/rename.cc:537`（Rename::renameInsts）
   - DISPATCH: `src/cpu/o3/iew.cc:881`（IEW::dispatchInsts；LSQ 插入 :1015/:1026，IQ 插入 :1086）
   - ISSUE: `src/cpu/o3/inst_queue.cc:868`（scheduleReadyInsts；insert :669；FU 选择 ~:944）
   - EXECUTE: `src/cpu/o3/iew.cc:1138`（IEW::executeInsts）→ `src/cpu/o3/dyn_inst.cc:358`（DynInst::execute）
   - WRITEBACK: `src/cpu/o3/iew.cc:1380`（IEW::writebackInsts）
   - COMMIT: `src/cpu/o3/commit.cc:1125`（Commit::commitHead；ROB 插入在 Commit::getInsts :1367 → rob.cc:194）
   - 访存侧线：LSQUnit::insert `lsq_unit.cc:338` · executeLoad :642 · executeStore :714 · read :1376 · write :1730 · completeDataAccess :108 · writeback :1136
   - 分支侧线：BAC::predict `src/cpu/o3/bac.cc:565` · FTQ::insert `ftq.cc:186` · 误预测检测 `iew.cc:1596` · 重定向 `fetch.cc:711`（Fetch::doSquash）
   （以上路径均相对 `CHAOS/gem5/`）
7. **CHAOS 注入器挂载语义**（命中谓词依据；写入 families.yaml 与总图）：

```python
# tools/o3_atlas/corpus.py
"""Enumerate the armv8-isa instruction corpus — the single coverage baseline.

All counts are measured (2026-10-16) against docs/cpu/armv8-isa/ and frozen
here; any drift in the corpus makes these tests fail loudly instead of
silently under-covering.
"""
from pathlib import Path

REPO = Path(__file__).resolve().parents[2]
CORPUS_DIR = REPO / "docs" / "cpu" / "armv8-isa"

INDEX_PAGES = {  # 15 个分类索引页（文件名 = 分类名）
    "Arithmetic", "Atomic", "Barrier-Hint", "Bit-Manipulation", "Branch",
    "Compare", "Conditional-Select", "Count", "Cryptographic", "Load-Store",
    "Logical", "Other", "PC-Relative", "Pointer-Authentication", "System",
}

EXPECTED_COUNTS = {  # 372 指令页的实测分账（2026-10-16）
    "Load-Store": 123, "Other": 81, "Arithmetic": 33, "Bit-Manipulation": 25,
    "Logical": 23, "System": 16, "Barrier-Hint": 16, "Branch": 13,
    "Pointer-Authentication": 11, "Compare": 11, "Conditional-Select": 7,
    "Atomic": 7, "PC-Relative": 2, "Cryptographic": 2, "Count": 2,
}

def corpus_files():
    return sorted(p for p in CORPUS_DIR.glob("*.md") if p.is_file())

def index_pages():
    return [p for p in corpus_files() if p.stem in INDEX_PAGES]

def instruction_pages():
    return [p for p in corpus_files() if p.stem not in INDEX_PAGES]

def category_of(path: Path) -> str:
    stem = Path(path).stem
    for cat in sorted(INDEX_PAGES, key=len, reverse=True):  # 长名优先，防前缀误配
        if stem.startswith(cat + "-"):
            return cat
    raise ValueError(f"unknown category for {path}")

def title_of(path: Path) -> str:
    for line in Path(path).read_text(encoding="utf-8", errors="replace").splitlines():
        if line.startswith("# "):
            return line[2:].strip()
    raise ValueError(f"no title heading in {path}")
```

```python
# tools/o3_atlas/schema.py
"""Hand-rolled record/family validator.

jsonschema is NOT installed on this host (pip offline) — same situation and
same rationale as tools/manifest_validate.py: this module IS the enforced
spec, kept honest by tests/test_schema.py.
"""
import re
from .corpus import instruction_pages

TIERS = ("implemented", "partial", "unimplemented", "fs-only", "macro-op")
BEHAVIORS = ("absent", "warn-nop", "illegal-fault")  # unimplemented 的三种真实形态
NODES = ("FETCH", "DECODE", "RENAME", "DISPATCH", "ISSUE", "EXECUTE", "MEMORY",
         "WRITEBACK", "COMMIT", "BPU", "FTQ", "REDIRECT")  # 标准节点词表
QUEUES = ("IQ", "LSQ")
INJECTORS = (
    "CHAOSDecode", "CHAOSRenameMap", "CHAOSFreeList", "CHAOSPhysReg", "CHAOSIQ",
    "CHAOSLSQFwd", "CHAOSL1DForward", "CHAOSAddrPath", "CHAOSROB", "CHAOSReg",
    "CHAOSExec", "CHAOSFPU", "CHAOSBPU", "CHAOSRAS", "CHAOSExMon", "CHAOSProbe",
    "CHAOSCommitTrace", "CHAOSMem", "CHAOSCache",
)
ANCHOR_RE = re.compile(r"^[A-Za-z0-9_./+-]+:\d+$")

def _corpus_names():
    return {p.name for p in instruction_pages()}

def _check_anchor_list(errors, container, where):
    for ev in container:
        a = ev.get("anchor", "")
        if not ANCHOR_RE.match(a):
            errors.append(f"{where}: bad anchor format: {a!r}")
        if not str(ev.get("claim", "")).strip():
            errors.append(f"{where}: evidence claim empty")

def validate_family(fam, name, families):
    errs = []
    if not str(fam.get("summary", "")).strip():
        errs.append(f"family {name}: summary missing")
    pipe = fam.get("pipeline", [])
    if not pipe:
        errs.append(f"family {name}: pipeline empty")
    for n in pipe:
        if n not in NODES:
            errs.append(f"family {name}: unknown node {n!r}")
    if fam.get("queue") not in QUEUES:
        errs.append(f"family {name}: queue must be one of {QUEUES}")
    if not str(fam.get("fu", "")).strip():
        errs.append(f"family {name}: fu missing")
    for inj in fam.get("default_injectors", []):
        if inj not in INJECTORS:
            errs.append(f"family {name}: unknown injector {inj!r}")
    for node, a in fam.get("anchors", {}).items():
        if node not in NODES:
            errs.append(f"family {name}: anchor for unknown node {node!r}")
        if not ANCHOR_RE.match(a):
            errs.append(f"family {name}: bad anchor {a!r}")
    return errs

def validate_instruction(rec, families):
    errs = []
    where = rec.get("doc", "<no doc>")
    if where not in _corpus_names():
        errs.append(f"{where}: doc is not a corpus instruction page")
    if not str(rec.get("mnemonic", "")).strip():
        errs.append(f"{where}: mnemonic missing")
    tier = rec.get("tier")
    if tier not in TIERS:
        errs.append(f"{where}: tier {tier!r} not in {TIERS}")
        return errs
    if tier == "unimplemented":
        if rec.get("behavior") not in BEHAVIORS:
            errs.append(f"{where}: unimplemented requires behavior in {BEHAVIORS}")
    else:
        fam = rec.get("family")
        if fam not in families:
            errs.append(f"{where}: family {fam!r} undefined")
        gem5 = rec.get("gem5") or {}
        for k in ("decode_mnemonic", "static_inst"):
            if not str(gem5.get(k, "")).strip():
                errs.append(f"{where}: gem5.{k} required for tier={tier}")
        dec = (gem5.get("anchors") or {}).get("decoder", "")
        if not ANCHOR_RE.match(dec):
            errs.append(f"{where}: gem5.anchors.decoder required for tier={tier}")
    for inj in rec.get("injectors", []):
        if inj not in INJECTORS:
            errs.append(f"{where}: unknown injector {inj!r}")
    if not rec.get("evidence"):
        errs.append(f"{where}: evidence required")
    _check_anchor_list(errs, rec.get("evidence", []), where)
    return errs

def validate_all(families, instructions):
    errs = []
    for name, fam in families.items():
        errs += validate_family(fam, name, families)
    for rec in instructions:
        errs += validate_instruction(rec, families)
```

```python
# tools/o3_atlas/validate.py
"""o3-atlas 全量校验器：schema + 锚点验真 + 覆盖 + 生成确定性（Task 6 补全）。"""
import sys
from pathlib import Path
import yaml
from . import schema
from .corpus import REPO, instruction_pages

DATA = REPO / "docs/cpu/o3-atlas/data"

def check_anchor(anchor, repo_root):
    path_s, _, line_s = anchor.rpartition(":")
    p = Path(repo_root) / path_s
    if not p.is_file():
        return f"anchor file missing: {anchor}"
    line = int(line_s)
    lines = p.read_text(encoding="utf-8", errors="replace").splitlines()
    if not (1 <= line <= len(lines)):
        return f"anchor line out of range: {anchor} (file has {len(lines)} lines)"
    if not lines[line - 1].strip():
        return f"anchor points at blank line: {anchor}"
    return None

def load_data():
    families = yaml.safe_load((DATA / "families.yaml").read_text()) or {}
    instructions = []
    for f in sorted((DATA / "instructions").glob("*.yaml")):
        instructions += yaml.safe_load(f.read_text()) or []
    return families, instructions

def iter_anchors(families, instructions):
    for name, fam in families.items():
        for node, a in (fam.get("anchors") or {}).items():
            yield f"family {name}.{node}", a
    for rec in instructions:
        for ev in rec.get("evidence", []):
            yield f"{rec.get('doc')}.evidence", ev["anchor"]
        for k, a in ((rec.get("gem5") or {}).get("anchors") or {}).items():
            yield f"{rec.get('doc')}.gem5.{k}", a

def main():
    families, instructions = load_data()
    errs = schema.validate_all(families, instructions)
    for where, a in iter_anchors(families, instructions):
        e = check_anchor(a, REPO)
        if e:
            errs.append(f"{where}: {e}")
    if errs:
        print("VALIDATION FAILED:")
        for e in errs:
            print(" -", e)
        sys.exit(1)
    print(f"OK: {len(instructions)} records, {len(families)} families, anchors verified")

if __name__ == "__main__":
    main()
```

```python
# tools/o3_atlas/lsp_query.py
"""clangd JSON-RPC client for precise anchor discovery (def/ref/hover).

Scope note: compile_commands.json covers C++ TU under CHAOS/gem5/src; the
.isa DSL and generated .inc files are NOT compilation units — read those
directly and verify anchors with validate.py instead.
"""
import argparse, json, os, subprocess, sys

class ClangdClient:
    def __init__(self, ccd, clangd_bin=os.path.expanduser("~/.local/bin/clangd")):
        self.proc = subprocess.Popen(
            [clangd_bin, f"--compile-commands-dir={ccd}", "--background-index",
             "--pch-storage=memory", "--limit-references=100"],
            stdin=subprocess.PIPE, stdout=subprocess.PIPE,
            stderr=subprocess.DEVNULL)
        self._id = 0

    def _send(self, obj):
        data = json.dumps(obj).encode()
        self.proc.stdin.write(b"Content-Length: %d\r\n\r\n" % len(data) + data)
        self.proc.stdin.flush()

    def _read(self):
        headers = {}
        while True:
            line = self.proc.stdout.readline()
            if not line:
                raise EOFError("clangd closed")
            if line in (b"\r\n", b"\n"):
                break
            k, _, v = line.decode().partition(":")
            headers[k.strip().lower()] = v.strip()
        return json.loads(self.proc.stdout.read(int(headers["content-length"])))

    def request(self, method, params):
        self._id += 1
        rid = self._id
        self._send({"jsonrpc": "2.0", "id": rid, "method": method, "params": params})
        while True:
            msg = self._read()          # 跳过 server 通知，只认同 id 应答
            if msg.get("id") == rid:
                return msg.get("result")

    def notify(self, method, params):
        self._send({"jsonrpc": "2.0", "method": method, "params": params})

    def initialize(self, root):
        self.request("initialize", {"processId": os.getpid(),
                                    "rootUri": "file://" + root, "capabilities": {}})
        self.notify("initialized", {})

    def open(self, path):
        text = open(path, encoding="utf-8", errors="replace").read()
        self.notify("textDocument/didOpen", {"textDocument": {
            "uri": "file://" + path, "languageId": "cpp", "version": 1, "text": text}})

    def definition(self, path, line, col):
        return self.request("textDocument/definition", {
            "textDocument": {"uri": "file://" + path},
            "position": to_lsp_pos(line, col)})

def to_lsp_pos(line_1b, col_1b):
    return {"line": line_1b - 1, "character": col_1b - 1}

def _fmt(locs):
    out = []
    for l in (locs if isinstance(locs, list) else [locs]):
        if not l:
            continue
        out.append({"file": l["uri"].removeprefix("file://"),
                    "line": l["range"]["start"]["line"] + 1,
                    "col": l["range"]["start"]["character"] + 1})
    return out

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("cmd", choices=["def", "ref", "hover"])
    ap.add_argument("loc", help="FILE:LINE:COL (1-based)")
    ap.add_argument("--ccd", default="/home/sdc/gem5-fi-ding/build/ARM")
    args = ap.parse_args()
    path, line, col = args.loc.rsplit(":", 2)
    path = os.path.abspath(path)
    c = ClangdClient(args.ccd)
    c.initialize(os.path.dirname(path))
    c.open(path)
    method = {"def": "textDocument/definition",
              "ref": "textDocument/references",
              "hover": "textDocument/hover"}[args.cmd]
    res = c.request(method, {"textDocument": {"uri": "file://" + path},
                             "position": to_lsp_pos(int(line), int(col)),
                             "context": {"includeDeclaration": True}})
    json.dump({"query": args.loc, "results": _fmt(res)}, sys.stdout, indent=2, ensure_ascii=False)
    print()

if __name__ == "__main__":
    main()
```
