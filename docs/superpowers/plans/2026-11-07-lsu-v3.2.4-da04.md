## Review Focus

1. **非 4 字节/非全 1 mask 的 Store** —— 期望完全不进入 DA04 eligible 流（attempted 计数可以增加，eligible 不得增加）。
2. **对齐/跨 cache line 的 4 字节 Store** —— 期望 mask 仍在 `Request::_byteEnable` 上被切片传递，地址与 size 不变，不触发 gem5 assert。
3. **同一地址被后续 Store 覆盖** —— 期望 Trace 诚实记录 `store_mem_write` 的证据，但不伪造 Load/Consumer 链。
4. **Load 从 Store Queue 转发（forwarding）** —— gem5 的转发 memcpy 不读 byte-enable；期望 Trace 明确记录 `byte_enable_not_applied_in_forward`，终态按真实架构结果判定，不得标成“内存写入生效”。
5. **目标 Store 被 Squash / 未 Commit** —— 期望终态 `SQUASHED`，不能因出现过 `memory_response` 或 `register_writeback` 就判 `MASKED`。

### Byte-enable 的真实语义

| 内容 | 事实 | 依据 |
|---|---|---|
| 实际类型/宽度 | `std::vector<bool> _byteEnable`，长度 = 访问 size（不是固定 4-bit） | `CHAOS/gem5/src/mem/request.hh:403`；`Request::setByteEnable` assert `be.size()==_size` `request.hh:714-720` |
| 生成点 | `LSQ::pushRequest` 新建请求时赋值；ARM ISA 不传 byte_enable，故按 size 构造全 1 | `CHAOS/gem5/src/cpu/o3/lsq.cc:830-835` |
| bit/lane 对应 | index 0 = 最低地址字节（packet `data[0]`，AArch64 小端 = 寄存器 bits[7:0]）；size=4 时正好是 4-bit mask，bit0=最低字节 | `Packet::writeData` 逐字节按 `getByteEnable()[i]` 写入 `packet.hh:1321-1336` |
| 传递 | `addReq` 对每个 fragment `Request::setByteEnable` | `lsq.cc:1116-1129`；`lsq.cc:1005`；`lsq.cc:833-834` |
| Store Queue 是否保留 | SQ 保存的是数据/请求对象；mask 在 `LSQRequest::_byteEnable`（及每 fragment 的 `Request`）上，Drain 时 `buildPackets` 用同一 Request | `lsq_unit.cc:869-990`；`lsq.cc:1270-1300` |
| 是否真正影响内存写入 | 是。清 bit0 后 `Packet::isMaskedWrite()` 为真，`BaseCache::updateBlockData → cpkt->writeDataToBlock` 只写 enabled 字节 | `packet.hh:1449-1453`、`1321-1336`；`mem/cache/base.cc:838-858` |
| Load 路径 | 转发路径不看 byte-enable（`lsq_unit.cc` 无 `byteEnable` 引用）；内存路径返回 cache 中的真实字节 | 源码 grep；`lsq_unit.cc:1557-1623`、`107-215` |

**结论：** gem5 的 Byte-enable 不是固定 4-bit，但对 `size==4` 的 Store 恰好就是 4 个 lane，bit0 对应地址最低字节。官方设计（`4B Load/Store`）在本轮被研究者收紧为 **Store-only**；实现必须按 size==4 + 全 1 过滤，且不得触碰 Load。

```cpp
#ifndef __CPU_O3_DA04_BYTE_ENABLE_HH__
#define __CPU_O3_DA04_BYTE_ENABLE_HH__

#include <cstdint>
#include <vector>

namespace gem5
{
namespace chaos_da04
{

// DA04 (LSU v3.2.4): the modelled mask is the 4-bit byte-enable of a 32-bit
// store. gem5 stores it as std::vector<bool>, one entry per byte:
//   index 0 = lowest-addressed byte (packet data[0], AArch64 LE bits [7:0])
//   index 3 = highest-addressed byte (bits [31:24])
// 0xF = {true,true,true,true}; clearing bit0 yields 0xE =
// {false,true,true,true}. No Load and no non-4-byte access is eligible.
static constexpr unsigned kMaskWidth = 4;

inline bool
eligible(unsigned size, const std::vector<bool> &be)
{
    if (size != kMaskWidth || be.size() != kMaskWidth)
        return false;
    for (bool b : be) {
        if (!b)
            return false;
    }
    return true;
}

inline uint64_t
maskOf(const std::vector<bool> &be)
{
    uint64_t m = 0;
    for (unsigned i = 0; i < be.size() && i < 64; ++i) {
        if (be[i])
            m |= (uint64_t)1 << i;
    }
    return m;
}

inline bool
clearLane0(std::vector<bool> &be)
{
    if (be.empty() || !be[0])
        return false;
    be[0] = false;
    return true;
}

} // namespace chaos_da04
} // namespace gem5

#endif // __CPU_O3_DA04_BYTE_ENABLE_HH__
```

```python
"""Terminal adjudication on a fault chain (LSU v3.2.4).

Rules are chain-scoped: only events that share the injection's ptag (or, when
ptag is absent, its dyn_inst_seq) can justify a verdict. A load that was
squashed before commit is SQUASHED even if it produced a response and a
register writeback.
"""
from __future__ import annotations

LOAD_SIDE = ("A01", "A03", "FW02")


def _chain(events, injection):
    """The fault chain = same ptag (injection/store/load/consumer) OR the
    fault instruction's own seq (commit/squash/lifecycle nodes carry ptag=0).
    A different dynamic instruction can only join through a ptag chain."""
    if injection is None:
        return []
    ptag = injection.get("ptag")
    seq = injection.get("dyn_inst_seq")
    out = []
    for e in events:
        if ptag and e.get("ptag") == ptag:
            out.append(e)
        elif seq is not None and e.get("dyn_inst_seq") == seq:
            out.append(e)
    return out


def adjudicate(events, *, model, observed_terminal, oracle_pass,
               trace_complete=True):
    injection = None
    for e in events:
        if e.get("event_type") in ("injection", "injection_nochange"):
            injection = e
    if observed_terminal in ("CRASH", "HANG", "ARCH_STATE_CORRUPTION",
                             "DETECTED_CONTAINED"):
        return observed_terminal, None
    if not trace_complete:
        return "UNRESOLVED", "TRACE_INCOMPLETE"
    if injection is None:
        return "UNRESOLVED", "NO_INJECTION_EVENT"
    seq = injection.get("dyn_inst_seq")
    chain = _chain(events, injection)
    own = [e for e in chain if e.get("dyn_inst_seq") == seq]
    committed = any(e.get("event_type") in ("commit", "errant_commit")
                    for e in own)
    squashed = any(e.get("event_type") == "squash" for e in own)
    if squashed and not committed:
        return "SQUASHED", "TARGET_SQUASHED_NO_COMMIT"
    if oracle_pass is False:
        return "SDC", "ORACLE_FAIL"
    if model in LOAD_SIDE:
        data_value = any(
            e.get("event_type") in ("memory_response", "register_writeback")
            and e.get("new_value") not in (None, "")
            for e in own)
        if not committed:
            return "UNRESOLVED", "TARGET_NOT_COMMITTED"
        if not data_value:
            return "UNRESOLVED", "UNRESOLVED_NO_DATA_VALUE_EVIDENCE"
        return "MASKED", "TARGET_COMMITTED_DATA_VALUE_PRESENT"
    # store-side (SQ01 / DA04)
    if not committed:
        return "UNRESOLVED", "STORE_NOT_COMMITTED"
    consumed = any(e.get("event_type") in ("consumer_commit", "errant_commit")
                   and e.get("dyn_inst_seq") != seq for e in chain)
    return "MASKED", ("STORE_COMMITTED_LANE_CONSUMED_ORACLE_PASS"
                      if consumed else "STORE_COMMITTED_LANE_NOT_READ")
```
